#!/usr/bin/env python3
"""
Crypto trading bot: 24/7 scanner, trend signals, risk controls, backtester.
Modes:  backtest | paper (default, fake money) | live (real money)
Install: pip install ccxt pandas numpy matplotlib requests
Usage:  python bot.py backtest --days 90
        python bot.py paper
        python bot.py live --i-understand
Kill switch: create a file named STOP in this folder, or let the automatic
daily-loss / drawdown limits trip. The bot closes positions and halts.
NOT financial advice. No bot guarantees profit. Start in paper mode.
"""
import os, sys, time, json, argparse, logging
from datetime import datetime, timezone
import numpy as np
import pandas as pd
import ccxt

# ------------------------- SETTINGS (edit these) -------------------------
CFG = dict(
    exchange="binance",            # any ccxt exchange id: kraken, coinbase, bybit...
    quote="USDT",
    symbols=["BTC/USDT", "ETH/USDT", "SOL/USDT", "BNB/USDT", "XRP/USDT"],
    timeframe="5m",
    scan_seconds=10,               # market rescan interval
    start_balance=1000.0,          # paper / backtest starting cash
    risk_per_trade=0.01,           # risk 1% of equity per trade
    stop_atr=2.0,                  # stop-loss  = entry - 2 x ATR
    take_atr=3.0,                  # take-profit = entry + 3 x ATR
    max_positions=3,
    max_alloc_per_trade=0.33,      # never put more than 33% of equity in one coin
    daily_loss_limit=0.03,         # halt if down 3% on the day
    max_drawdown=0.10,             # halt if 10% below peak equity
    fee=0.001,                     # 0.1% per side
    slippage=0.0005,               # 0.05% assumed slippage
)
STATE_FILE, TRADE_LOG, STOP_FILE = "state.json", "trades.csv", "STOP"
logging.basicConfig(level=logging.INFO, format="%(asctime)s %(message)s",
                    handlers=[logging.StreamHandler(), logging.FileHandler("bot.log")])
log = logging.info

# ------------------------- INDICATORS & SIGNALS -------------------------
def add_signals(df):
    c = df.close
    df["ema_f"] = c.ewm(span=20, adjust=False).mean()
    df["ema_s"] = c.ewm(span=50, adjust=False).mean()
    d = c.diff()
    up = d.clip(lower=0).ewm(alpha=1/14, adjust=False).mean()
    dn = (-d.clip(upper=0)).ewm(alpha=1/14, adjust=False).mean()
    df["rsi"] = 100 - 100 / (1 + up / dn.replace(0, np.nan))
    macd = c.ewm(span=12, adjust=False).mean() - c.ewm(span=26, adjust=False).mean()
    df["macd_h"] = macd - macd.ewm(span=9, adjust=False).mean()
    tr = pd.concat([df.high - df.low, (df.high - c.shift()).abs(),
                    (df.low - c.shift()).abs()], axis=1).max(axis=1)
    df["atr"] = tr.ewm(alpha=1/14, adjust=False).mean()
    up_, dn_ = df.high.diff(), -df.low.diff()
    pdm = pd.Series(np.where((up_ > dn_) & (up_ > 0), up_, 0.0), index=df.index)
    mdm = pd.Series(np.where((dn_ > up_) & (dn_ > 0), dn_, 0.0), index=df.index)
    pdi = 100 * pdm.ewm(alpha=1/14, adjust=False).mean() / df.atr
    mdi = 100 * mdm.ewm(alpha=1/14, adjust=False).mean() / df.atr
    dx = 100 * (pdi - mdi).abs() / (pdi + mdi).replace(0, np.nan)
    df["adx"] = dx.ewm(alpha=1/14, adjust=False).mean()   # trend strength (filters chop)
    df["ema200"] = c.ewm(span=200, adjust=False).mean()    # long-term trend filter
    df["vol_ratio"] = df.volume / df.volume.rolling(20).mean()
    df["trend"] = np.where(df.ema_f > df.ema_s, "UP", "DOWN")
    ok = ((df.ema_f > df.ema_s) & (c > df.ema_s) & (df.macd_h > 0) &
          (df.macd_h > df.macd_h.shift()) & df.rsi.between(50, 70) & (df.vol_ratio > 1) &
          (df.adx > 20) & (c > df.ema200))
    df["buy"] = ok & ~ok.shift(fill_value=False)       # fresh entry signal only
    df["sell"] = (df.ema_f < df.ema_s) | (df.rsi > 80)  # trend flip / overheated
    return df

def to_df(raw):
    df = pd.DataFrame(raw, columns=["ts", "open", "high", "low", "close", "volume"])
    df = df.drop_duplicates("ts").reset_index(drop=True)
    df["time"] = pd.to_datetime(df.ts, unit="ms")
    return add_signals(df)

def make_exchange(keys=False):
    p = {"enableRateLimit": True}
    if keys:
        p.update(apiKey=os.environ["API_KEY"], secret=os.environ["API_SECRET"])
    return getattr(ccxt, CFG["exchange"])(p)

def notify(msg):
    log(msg)
    t, c = os.getenv("TELEGRAM_TOKEN"), os.getenv("TELEGRAM_CHAT_ID")
    if t and c:
        try:
            import requests
            requests.post(f"https://api.telegram.org/bot{t}/sendMessage",
                          data={"chat_id": c, "text": msg}, timeout=5)
        except Exception as e:
            log(f"telegram failed: {e}")

# ------------------------- BACKTEST -------------------------
def fetch_history(ex, sym, tf, days):
    step = ex.parse_timeframe(tf) * 1000
    since, out = ex.milliseconds() - days * 86400000, []
    while since < ex.milliseconds():
        b = ex.fetch_ohlcv(sym, tf, since=since, limit=1000)
        if not b: break
        out += b
        since = b[-1][0] + step
        if len(b) < 2: break
    return to_df(out)

def backtest_symbol(df):
    cash, pos, pending, trades, eq = CFG["start_balance"], None, False, [], []
    f, s = CFG["fee"], CFG["slippage"]
    cool = 0
    for i in range(200, len(df)):
        r = df.iloc[i]
        if pos is None and pending and not np.isnan(r.atr):
            entry = r.open * (1 + s)
            stop, tp = entry - CFG["stop_atr"] * pos_atr, entry + CFG["take_atr"] * pos_atr
            risk_cash = cash * CFG["risk_per_trade"]
            qty = min(risk_cash / (entry - stop), cash * CFG["max_alloc_per_trade"] / entry)
            if qty > 0:
                cash -= qty * entry * (1 + f)
                pos = dict(qty=qty, entry=entry, stop=stop, tp=tp, t=r.time, a=pos_atr)
        pending = False
        if pos:
            exit_p = None
            if r.low <= pos["stop"]: exit_p, why = pos["stop"], "stop"
            elif r.high >= pos["tp"]: exit_p, why = pos["tp"], "target"
            elif r.sell: exit_p, why = r.close, "trend flip"
            if exit_p:
                exit_p *= (1 - s)
                cash += pos["qty"] * exit_p * (1 - f)
                trades.append(dict(open=pos["t"], close=r.time, entry=pos["entry"], exit=exit_p,
                                   pnl_pct=(exit_p / pos["entry"] - 1) * 100, why=why))
                if exit_p < pos["entry"]: cool = i + 6   # 30-min cooldown after a loss
                pos = None
            elif r.high - pos["entry"] >= pos["a"]:   # breakeven + trailing stop
                pos["stop"] = max(pos["stop"], r.high - 1.5 * pos["a"], pos["entry"] * (1 + 2 * f))
        elif r.buy and i >= cool:
            pending, pos_atr = True, r.atr
        eq.append(cash + (pos["qty"] * r.close if pos else 0))
    return pd.DataFrame(trades), pd.Series(eq, index=df.time[200:])

def run_backtest(days):
    import matplotlib; matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    ex = make_exchange()
    for sym in CFG["symbols"]:
        df = fetch_history(ex, sym, CFG["timeframe"], days)
        tr, eq = backtest_symbol(df)
        ret = (eq.iloc[-1] / CFG["start_balance"] - 1) * 100
        hold = (df.close.iloc[-1] / df.close.iloc[200] - 1) * 100
        dd = ((eq / eq.cummax()) - 1).min() * 100
        wins = (tr.pnl_pct > 0).mean() * 100 if len(tr) else 0
        gp = tr.pnl_pct[tr.pnl_pct > 0].sum() if len(tr) else 0
        gl = -tr.pnl_pct[tr.pnl_pct < 0].sum() if len(tr) else 0
        notify(f"{sym} {days}d | return {ret:+.1f}% vs buy&hold {hold:+.1f}% | trades {len(tr)} | "
               f"win {wins:.0f}% | profit factor {gp / gl if gl else float('inf'):.2f} | max DD {dd:.1f}%")
        fig, ax = plt.subplots(2, 1, figsize=(14, 8), sharex=True, gridspec_kw={"height_ratios": [3, 1]})
        ax[0].plot(df.time, df.close, lw=.8, label="price")
        ax[0].plot(df.time, df.ema_f, lw=.6, label="EMA20"); ax[0].plot(df.time, df.ema_s, lw=.6, label="EMA50")
        if len(tr):
            ax[0].scatter(tr.open, tr.entry, marker="^", c="g", s=40, label="buy")
            ax[0].scatter(tr.close, tr.exit, marker="v", c="r", s=40, label="sell")
        ax[0].legend(); ax[0].set_title(f"{sym} {CFG['timeframe']}  return {ret:+.1f}%")
        ax[1].plot(eq.index, eq.values); ax[1].set_title("Equity")
        fig.tight_layout(); fig.savefig(f"backtest_{sym.replace('/', '_')}.png", dpi=110); plt.close(fig)
        tr.to_csv(f"backtest_{sym.replace('/', '_')}_trades.csv", index=False)

# ------------------------- LIVE / PAPER ENGINE -------------------------
def load_state():
    if os.path.exists(STATE_FILE):
        return json.load(open(STATE_FILE))
    return dict(cash=CFG["start_balance"], pos={}, peak=CFG["start_balance"],
                day=str(datetime.now(timezone.utc).date()), day_start=CFG["start_balance"], halted=False)

def save_state(st): json.dump(st, open(STATE_FILE, "w"), indent=1)

def log_trade(row):
    pd.DataFrame([row]).to_csv(TRADE_LOG, mode="a", header=not os.path.exists(TRADE_LOG), index=False)

def run_bot(live):
    ex = make_exchange(keys=live)
    ex.load_markets()
    st = load_state()
    notify(f"Bot started in {'LIVE' if live else 'PAPER'} mode on {CFG['exchange']}")
    f = CFG["fee"]

    def equity(prices):
        return st["cash"] + sum(p["qty"] * prices.get(s, p["entry"]) for s, p in st["pos"].items())

    def close_pos(sym, price, why):
        p = st["pos"].pop(sym)
        if live:
            ex.create_market_sell_order(sym, ex.amount_to_precision(sym, p["qty"]))
        st["cash"] += p["qty"] * price * (1 - f)
        pnl = (price / p["entry"] - 1) * 100
        if pnl < 0: st.setdefault("cool", {})[sym] = time.time() + 1800
        log_trade(dict(time=datetime.now(timezone.utc).isoformat(), symbol=sym, side="SELL",
                       price=price, qty=p["qty"], pnl_pct=round(pnl, 2), reason=why))
        notify(f"SELL {sym} @ {price:.4f} ({pnl:+.2f}%) - {why}")

    while True:
        try:
            if live:
                st["cash"] = ex.fetch_balance()["free"].get(CFG["quote"], 0)
            today = str(datetime.now(timezone.utc).date())
            data, prices = {}, {}
            for sym in CFG["symbols"]:
                data[sym] = to_df(ex.fetch_ohlcv(sym, CFG["timeframe"], limit=300))
                prices[sym] = float(data[sym].close.iloc[-1])
            eq = equity(prices)
            if today != st["day"]:
                st["day"], st["day_start"] = today, eq
            st["peak"] = max(st["peak"], eq)

            # ---- automatic kill switch ----
            reason = None
            if os.path.exists(STOP_FILE): reason = "STOP file found"
            elif eq < st["day_start"] * (1 - CFG["daily_loss_limit"]): reason = "daily loss limit hit"
            elif eq < st["peak"] * (1 - CFG["max_drawdown"]): reason = "max drawdown hit"
            if reason and not st["halted"]:
                for s in list(st["pos"]): close_pos(s, prices[s], f"KILL SWITCH: {reason}")
                st["halted"] = True
                notify(f"BOT HALTED: {reason}. Equity {eq:.2f}. Delete state.json/STOP to reset.")
            if st["halted"]:
                save_state(st); log("halted; exiting"); return

            # ---- manage open positions (check live price vs stop/target) ----
            for s in list(st["pos"]):
                p, px, last = st["pos"][s], prices[s], data[s].iloc[-2]
                if px <= p["stop"]: close_pos(s, px, "stop-loss")
                elif px >= p["tp"]: close_pos(s, px, "take-profit")
                elif last.sell: close_pos(s, px, "trend flip / RSI>80")
                elif px - p["entry"] >= p.get("a", 0) > 0:
                    p["stop"] = max(p["stop"], px - 1.5 * p["a"], p["entry"] * (1 + 2 * f))

            # ---- look for entries on the last CLOSED candle ----
            for sym in CFG["symbols"]:
                last = data[sym].iloc[-2]
                if (sym in st["pos"] or len(st["pos"]) >= CFG["max_positions"] or not last.buy
                        or st.setdefault("last", {}).get(sym) == int(last.ts)
                        or st.get("cool", {}).get(sym, 0) > time.time()):
                    continue
                px = prices[sym] * (1 + CFG["slippage"])
                stop, tp = px - CFG["stop_atr"] * last.atr, px + CFG["take_atr"] * last.atr
                qty = min(eq * CFG["risk_per_trade"] / (px - stop),
                          eq * CFG["max_alloc_per_trade"] / px, st["cash"] / (px * (1 + f)))
                if qty * px < 10: continue
                if live:
                    qty = float(ex.amount_to_precision(sym, qty))
                    ex.create_market_buy_order(sym, qty)
                st["cash"] -= qty * px * (1 + f)
                st["pos"][sym] = dict(qty=qty, entry=px, stop=stop, tp=tp, a=float(last.atr))
                st["last"][sym] = int(last.ts)  # don't re-enter on the same candle
                log_trade(dict(time=datetime.now(timezone.utc).isoformat(), symbol=sym, side="BUY",
                               price=px, qty=qty, pnl_pct="", reason="signal"))
                notify(f"BUY {sym} @ {px:.4f} | stop {stop:.4f} | target {tp:.4f}")

            save_state(st)
            table = " | ".join(f"{s.split('/')[0]} {prices[s]:.2f} {data[s].trend.iloc[-1]} "
                               f"RSI{data[s].rsi.iloc[-1]:.0f}" for s in CFG["symbols"])
            log(f"EQ {eq:.2f} (day {eq / st['day_start'] * 100 - 100:+.2f}%) pos {list(st['pos'])} || {table}")
        except (ccxt.NetworkError, ccxt.ExchangeNotAvailable) as e:
            log(f"network issue, retrying: {e}")
        except Exception as e:
            notify(f"ERROR: {e}")
        time.sleep(CFG["scan_seconds"])

# ------------------------- ENTRY POINT -------------------------
if __name__ == "__main__":
    ap = argparse.ArgumentParser()

function close(s,p,why){
 const q=S.pos[s];S.cash+=q.q*p*(1-cfg.fee);delete S.pos[s];S.cool=S.cool||{};if(p<q.entry)S.cool[s]=Date.now()+18e5;
 notify(`SELL ${s} @ ${p.toFixed(4)} (${((p/q.entry-1)*100).toFixed(2)}%) - ${why}`)}
async function tick(){
 if(busy)return;busy=true;
 try{
  await Promise.all(SYMS.map(async s=>{const k=ind(await kl(s,300));DATA[s]=k;PX[s]=k[k.length-1].c}));
  let eq=S.cash;for(const s in S.pos)eq+=S.pos[s].q*PX[s];
  if(today()!==S.day){S.day=today();S.dayStart=eq}
  S.peak=Math.max(S.peak,eq);
  const why=killReq?"manual kill switch":eq<S.dayStart*(1-cfg.dayLoss)?"daily loss limit":eq<S.peak*(1-cfg.maxDD)?"max drawdown":"";
  if(why){for(const s of Object.keys(S.pos))close(s,PX[s],"KILL: "+why);S.halted=true;notify("BOT HALTED: "+why);stop(true)}
  else{
   for(const s of Object.keys(S.pos)){const p=S.pos[s],d=DATA[s],px=PX[s];
    if(px<=p.stop)close(s,px,p.stop>p.entry?"trailing stop":"stop-loss");else if(px>=p.tp)close(s,px,"take-profit");else if(d[d.length-2].sell)close(s,px,"trend flip");
    else if(p.a&&px-p.entry>=p.a)p.stop=Math.max(p.stop,px-1.5*p.a,p.entry*(1+2*cfg.fee))}
   for(const s of SYMS){const d=DATA[s],r=d[d.length-2];
    if(S.pos[s]||Object.keys(S.pos).length>=cfg.maxPos||!r.buy||S.last[s]===r.t||(S.cool&&S.cool[s]>Date.now()))continue;
    const px=PX[s]*(1+cfg.slip),stop=px-cfg.stopAtr*r.atr,tp=px+cfg.tpAtr*r.atr;
    const q=Math.min(eq*cfg.risk/(px-stop),eq*cfg.alloc/px,S.cash/(px*(1+cfg.fee)));
    if(q*px<10)continue;S.cash-=q*px*(1+cfg.fee);S.pos[s]={q,entry:px,stop,tp,a:r.atr};S.last[s]=r.t;
    notify(`BUY ${s} @ ${px.toFixed(4)} | stop ${stop.toFixed(4)} | target ${tp.toFixed(4)}`)}}
  sv("s",S);$("st").textContent=S.halted?"Halted. Reset account to restart.":"Scanning every 10s · "+new Date().toLocaleTimeString();
  render(eq)
 }catch(e){$("st").textContent="Network issue, retrying: "+e.message}
 busy=false}

async function hist(s){let all=[],end;for(let i=0;i<3;i++){const k=await kl(s,1000,end);if(!k.length)break;all=k.concat(all);end=k[0].t-1}return ind(all)}
function bt(d){let cash=1000,pos=null,pend=false,pa=0,tr=[],pk=1000,dd=0,cool=0;
 for(let i=200;i<d.length;i++){const r=d[i];
  if(!pos&&pend){const e=r.o*(1+cfg.slip),st=e-cfg.stopAtr*pa,tp=e+cfg.tpAtr*pa,q=Math.min(cash*cfg.risk/(e-st),cash*cfg.alloc/e);if(q>0){cash-=q*e*(1+cfg.fee);pos={q,e,st,tp,a:pa}}}
  pend=false;
  if(pos){let x=0;if(r.l<=pos.st)x=pos.st;else if(r.h>=pos.tp)x=pos.tp;else if(r.sell)x=r.c;
   if(x){x*=1-cfg.slip;cash+=pos.q*x*(1-cfg.fee);tr.push(x/pos.e-1);if(x<pos.e)cool=i+6;pos=null}
   else if(r.h-pos.e>=pos.a)pos.st=Math.max(pos.st,r.h-1.5*pos.a,pos.e*(1+2*cfg.fee))}
  else if(r.buy&&i>=cool){pend=true;pa=r.atr}
  const eq=cash+(pos?pos.q*r.c:0);pk=Math.max(pk,eq);dd=Math.min(dd,eq/pk-1)}
 const last=d[d.length-1],eq=cash+(pos?pos.q*last.c:0),w=tr.filter(x=>x>0).length;
 return{ret:(eq/1000-1)*100,hold:(last.c/d[200].c-1)*100,n:tr.length,win:tr.length?w/tr.length*100:0,dd:dd*100}}

 $("btr").innerHTML='<span class="mu">Running… about 15 seconds.</span>';

  $("btr").innerHTML=out+'<small class="mu">Past results do not predict future results. Judge it against buy &amp; hold.</small>'}
 catch(e){$("btr").innerHTML='<span class="dn">Could not load history: '+e.message+'</span>'}}
