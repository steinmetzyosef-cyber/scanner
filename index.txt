#!/usr/bin/env python3
"""
Crypto trading bot: 24/7 scanner, trend signals, risk controls, backtester.
Modes:  backtest | paper (default, fake money) | live (real money)
Install: pip install ccxt pandas numpy matplotlib requests
Usage:  python bot.py backtest --days 90
        python bot.py paper
        python bot.py live --i-understand
Kill switch: create a file named STOP in this folder, or let the automatic
daily-loss / drawdown limits trip. The bot closes positions and halts.
NOT financial advice. No bot guarantees profit. Start in paper mode.
"""
import os, sys, time, json, argparse, logging
from datetime import datetime, timezone
import numpy as np
import pandas as pd
import ccxt

# ------------------------- SETTINGS (edit these) -------------------------
CFG = dict(
    exchange="binance",            # any ccxt exchange id: kraken, coinbase, bybit...
    quote="USDT",
    symbols=["BTC/USDT", "ETH/USDT", "SOL/USDT", "BNB/USDT", "XRP/USDT"],
    timeframe="5m",
    scan_seconds=10,               # market rescan interval
    start_balance=1000.0,          # paper / backtest starting cash
    risk_per_trade=0.01,           # risk 1% of equity per trade
    stop_atr=2.0,                  # stop-loss  = entry - 2 x ATR
    take_atr=3.0,                  # take-profit = entry + 3 x ATR
    max_positions=3,
    max_alloc_per_trade=0.33,      # never put more than 33% of equity in one coin
    daily_loss_limit=0.03,         # halt if down 3% on the day
    max_drawdown=0.10,             # halt if 10% below peak equity
    fee=0.001,                     # 0.1% per side
    slippage=0.0005,               # 0.05% assumed slippage
)
STATE_FILE, TRADE_LOG, STOP_FILE = "state.json", "trades.csv", "STOP"
logging.basicConfig(level=logging.INFO, format="%(asctime)s %(message)s",
                    handlers=[logging.StreamHandler(), logging.FileHandler("bot.log")])
log = logging.info

# ------------------------- INDICATORS & SIGNALS -------------------------
def add_signals(df):
    c = df.close
    df["ema_f"] = c.ewm(span=20, adjust=False).mean()
    df["ema_s"] = c.ewm(span=50, adjust=False).mean()
    d = c.diff()
    up = d.clip(lower=0).ewm(alpha=1/14, adjust=False).mean()
    dn = (-d.clip(upper=0)).ewm(alpha=1/14, adjust=False).mean()
    df["rsi"] = 100 - 100 / (1 + up / dn.replace(0, np.nan))
    macd = c.ewm(span=12, adjust=False).mean() - c.ewm(span=26, adjust=False).mean()
    df["macd_h"] = macd - macd.ewm(span=9, adjust=False).mean()
    tr = pd.concat([df.high - df.low, (df.high - c.shift()).abs(),
                    (df.low - c.shift()).abs()], axis=1).max(axis=1)
    df["atr"] = tr.ewm(alpha=1/14, adjust=False).mean()
    up_, dn_ = df.high.diff(), -df.low.diff()
    pdm = pd.Series(np.where((up_ > dn_) & (up_ > 0), up_, 0.0), index=df.index)
    mdm = pd.Series(np.where((dn_ > up_) & (dn_ > 0), dn_, 0.0), index=df.index)
    pdi = 100 * pdm.ewm(alpha=1/14, adjust=False).mean() / df.atr
    mdi = 100 * mdm.ewm(alpha=1/14, adjust=False).mean() / df.atr
    dx = 100 * (pdi - mdi).abs() / (pdi + mdi).replace(0, np.nan)
    df["adx"] = dx.ewm(alpha=1/14, adjust=False).mean()   # trend strength (filters chop)
    df["ema200"] = c.ewm(span=200, adjust=False).mean()    # long-term trend filter
    df["vol_ratio"] = df.volume / df.volume.rolling(20).mean()
    df["trend"] = np.where(df.ema_f > df.ema_s, "UP", "DOWN")
    ok = ((df.ema_f > df.ema_s) & (c > df.ema_s) & (df.macd_h > 0) &
          (df.macd_h > df.macd_h.shift()) & df.rsi.between(50, 70) & (df.vol_ratio > 1) &
          (df.adx > 20) & (c > df.ema200))
    df["buy"] = ok & ~ok.shift(fill_value=False)       # fresh entry signal only
    df["sell"] = (df.ema_f < df.ema_s) | (df.rsi > 80)  # trend flip / overheated
    return df

def to_df(raw):
    df = pd.DataFrame(raw, columns=["ts", "open", "high", "low", "close", "volume"])
    df = df.drop_duplicates("ts").reset_index(drop=True)
    df["time"] = pd.to_datetime(df.ts, unit="ms")
    return add_signals(df)

def make_exchange(keys=False):
    p = {"enableRateLimit": True}
    if keys:
        p.update(apiKey=os.environ["API_KEY"], secret=os.environ["API_SECRET"])
    return getattr(ccxt, CFG["exchange"])(p)

def notify(msg):
    log(msg)
    t, c = os.getenv("TELEGRAM_TOKEN"), os.getenv("TELEGRAM_CHAT_ID")
    if t and c:
        try:
            import requests
            requests.post(f"https://api.telegram.org/bot{t}/sendMessage",
                          data={"chat_id": c, "text": msg}, timeout=5)
        except Exception as e:
            log(f"telegram failed: {e}")

# ------------------------- BACKTEST -------------------------
def fetch_history(ex, sym, tf, days):
    step = ex.parse_timeframe(tf) * 1000
    since, out = ex.milliseconds() - days * 86400000, []
    while since < ex.milliseconds():
        b = ex.fetch_ohlcv(sym, tf, since=since, limit=1000)
        if not b: break
        out += b
        since = b[-1][0] + step
        if len(b) < 2: break
    return to_df(out)

def backtest_symbol(df):
    cash, pos, pending, trades, eq = CFG["start_balance"], None, False, [], []
    f, s = CFG["fee"], CFG["slippage"]
    cool = 0
    for i in range(200, len(df)):
        r = df.iloc[i]
        if pos is None and pending and not np.isnan(r.atr):
            entry = r.open * (1 + s)
            stop, tp = entry - CFG["stop_atr"] * pos_atr, entry + CFG["take_atr"] * pos_atr
            risk_cash = cash * CFG["risk_per_trade"]
            qty = min(risk_cash / (entry - stop), cash * CFG["max_alloc_per_trade"] / entry)
            if qty > 0:
                cash -= qty * entry * (1 + f)
                pos = dict(qty=qty, entry=entry, stop=stop, tp=tp, t=r.time, a=pos_atr)
        pending = False
        if pos:
            exit_p = None
            if r.low <= pos["stop"]: exit_p, why = pos["stop"], "stop"
            elif r.high >= pos["tp"]: exit_p, why = pos["tp"], "target"
            elif r.sell: exit_p, why = r.close, "trend flip"
            if exit_p:
                exit_p *= (1 - s)
                cash += pos["qty"] * exit_p * (1 - f)
                trades.append(dict(open=pos["t"], close=r.time, entry=pos["entry"], exit=exit_p,
                                   pnl_pct=(exit_p / pos["entry"] - 1) * 100, why=why))
                if exit_p < pos["entry"]: cool = i + 6   # 30-min cooldown after a loss
                pos = None
            elif r.high - pos["entry"] >= pos["a"]:   # breakeven + trailing stop
                pos["stop"] = max(pos["stop"], r.high - 1.5 * pos["a"], pos["entry"] * (1 + 2 * f))
        elif r.buy and i >= cool:
            pending, pos_atr = True, r.atr
        eq.append(cash + (pos["qty"] * r.close if pos else 0))
    return pd.DataFrame(trades), pd.Series(eq, index=df.time[200:])

def run_backtest(days):
    import matplotlib; matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    ex = make_exchange()
    for sym in CFG["symbols"]:
        df = fetch_history(ex, sym, CFG["timeframe"], days)
        tr, eq = backtest_symbol(df)
        ret = (eq.iloc[-1] / CFG["start_balance"] - 1) * 100
        hold = (df.close.iloc[-1] / df.close.iloc[200] - 1) * 100
        dd = ((eq / eq.cummax()) - 1).min() * 100
        wins = (tr.pnl_pct > 0).mean() * 100 if len(tr) else 0
        gp = tr.pnl_pct[tr.pnl_pct > 0].sum() if len(tr) else 0
        gl = -tr.pnl_pct[tr.pnl_pct < 0].sum() if len(tr) else 0
        notify(f"{sym} {days}d | return {ret:+.1f}% vs buy&hold {hold:+.1f}% | trades {len(tr)} | "
               f"win {wins:.0f}% | profit factor {gp / gl if gl else float('inf'):.2f} | max DD {dd:.1f}%")
        fig, ax = plt.subplots(2, 1, figsize=(14, 8), sharex=True, gridspec_kw={"height_ratios": [3, 1]})
        ax[0].plot(df.time, df.close, lw=.8, label="price")
        ax[0].plot(df.time, df.ema_f, lw=.6, label="EMA20"); ax[0].plot(df.time, df.ema_s, lw=.6, label="EMA50")
        if len(tr):
            ax[0].scatter(tr.open, tr.entry, marker="^", c="g", s=40, label="buy")
            ax[0].scatter(tr.close, tr.exit, marker="v", c="r", s=40, label="sell")
        ax[0].legend(); ax[0].set_title(f"{sym} {CFG['timeframe']}  return {ret:+.1f}%")
        ax[1].plot(eq.index, eq.values); ax[1].set_title("Equity")
        fig.tight_layout(); fig.savefig(f"backtest_{sym.replace('/', '_')}.png", dpi=110); plt.close(fig)
        tr.to_csv(f"backtest_{sym.replace('/', '_')}_trades.csv", index=False)

# ------------------------- LIVE / PAPER ENGINE -------------------------
def load_state():
    if os.path.exists(STATE_FILE):
        return json.load(open(STATE_FILE))
    return dict(cash=CFG["start_balance"], pos={}, peak=CFG["start_balance"],
                day=str(datetime.now(timezone.utc).date()), day_start=CFG["start_balance"], halted=False)

def save_state(st): json.dump(st, open(STATE_FILE, "w"), indent=1)

def log_trade(row):
    pd.DataFrame([row]).to_csv(TRADE_LOG, mode="a", header=not os.path.exists(TRADE_LOG), index=False)

def run_bot(live):
    ex = make_exchange(keys=live)
    ex.load_markets()
    st = load_state()
    notify(f"Bot started in {'LIVE' if live else 'PAPER'} mode on {CFG['exchange']}")
    f = CFG["fee"]

    def equity(prices):
        return st["cash"] + sum(p["qty"] * prices.get(s, p["entry"]) for s, p in st["pos"].items())

    def close_pos(sym, price, why):
        p = st["pos"].pop(sym)
        if live:
